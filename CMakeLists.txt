cmake_minimum_required(VERSION 3.21)
project(mnTransfer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network WebSockets Widgets Test)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY NAMES sodium libsodium REQUIRED)
add_library(transfer-core src/crypto.cpp src/transfer.cpp src/relay.cpp)
target_include_directories(transfer-core PUBLIC src ${SODIUM_INCLUDE_DIR})
target_link_libraries(transfer-core PUBLIC Qt6::Core Qt6::Network Qt6::WebSockets ${SODIUM_LIBRARY})
if(WIN32)
  target_compile_definitions(transfer-core PUBLIC SODIUM_STATIC)
  target_link_libraries(transfer-core PUBLIC bcrypt advapi32 ws2_32)
endif()
if(MSVC)
  target_compile_options(transfer-core PUBLIC /utf-8 /W4)
else()
  target_compile_options(transfer-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(mnTransfer src/main.cpp src/window.cpp)
target_link_libraries(mnTransfer PRIVATE transfer-core Qt6::Widgets)
set_target_properties(mnTransfer PROPERTIES MACOSX_BUNDLE TRUE)
if(WIN32)
  set_target_properties(mnTransfer PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
enable_testing()
add_executable(transfer-tests tests/transfer_tests.cpp)
target_link_libraries(transfer-tests PRIVATE transfer-core Qt6::Test)
add_dependencies(transfer-tests mnTransfer)
add_test(NAME transfer-tests COMMAND transfer-tests)
set_tests_properties(transfer-tests PROPERTIES TIMEOUT 180)
install(TARGETS mnTransfer RUNTIME DESTINATION . BUNDLE DESTINATION .)
